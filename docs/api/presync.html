presync

presync - recover trustworthy timing from recordings of the PRE-Sync signal.

The generator emits a pseudo-random square wave whose every transition is determined by (seed, configuration). That makes the emitted waveform reproducible offline, so it can be regenerated and used as ground truth: each recorded edge has a known true time, and the difference is the recorder's error.

The pipeline, in the order a full analysis runs them:

load -> locate -> score -> chunk -> judge -> compare -> report

load recording file -> edge times (XDF, C3D, CSV) locate find where the recording sits in the generator's sequence score offset, drift, jitter, capture, loss -- against the template chunk per-window quality, on timecode frames where present judge turn the numbers into findings a person can act on compare streams against each other, once each is scored on its own report text and PDF

One call runs all of it:

presync.analyze_file("session.xdf") # -> Analysis python -m presync run session.xdf --pdf out.pdf

This package is an organisation of existing, verified code. The analysis functions live in the modules alongside it (truth, align, analyze, edge_sync, timecode, diagnose) and are re-exported here so there is one obvious entry point per stage rather than five overlapping ones.

421def load_xdf(path, verbose=False):
422    """Every sync-bearing stream in an XDF file."""
423    import pyxdf
424    streams, _ = pyxdf.load_xdf(path, dejitter_timestamps=False,
425                                synchronize_clocks=False)
426    out = []
427    for s in streams:
428        info = s["info"]
429        name = info["name"][0]
430        ts = np.asarray(s["time_stamps"], float)
431        d = np.asarray(s["time_series"])
432        if d.dtype.kind not in "fiu" or d.ndim != 2 or len(ts) < 20:
433            continue
434        labels = []
435        try:
436            labels = [c["label"][0] for c in
437                      info["desc"][0]["channels"][0]["channel"]]
438        except Exception:
439            pass
440        clk = None
441        try:
442            v = [float(x["value"][0]) for x in
443                 s["footer"]["info"]["clock_offsets"][0]["offset"]]
444            if v:
445                clk = float(np.median(v))
446        except Exception:
447            pass
448
449        best = None
450        for ci in range(d.shape[1]):
451            col = d[:, ci].astype(float)
452            if not np.isfinite(col).all() or col.std() == 0:
453                continue
454            lab = labels[ci] if ci < len(labels) else f"ch{ci}"
455            lo, hi = col.min(), col.max()
456            mid = (lo + hi) / 2
457            up = np.flatnonzero((col[1:] > mid) & (col[:-1] <= mid)) + 1
458            dn = np.flatnonzero((col[1:] <= mid) & (col[:-1] > mid)) + 1
459            if len(up) < 5:
460                continue
461            is_trig = "trig" in lab.lower()
462            # A trigger line pulses briefly, so only its rises are events; an
463            # analog channel carries the whole wave, so both edges are real
464            # and using rises alone doubles every interval.
465            times = ts[up] if is_trig else ts[np.sort(np.concatenate([up, dn]))]
466            frac_ext = np.mean((col < lo + 0.1*(hi-lo)) | (col > hi - 0.1*(hi-lo)))
467            score_ = frac_ext + (1.0 if is_trig else 0.0)
468            if best is None or score_ > best[0]:
469                best = (score_, lab, times, is_trig)
470        if best is None:
471            continue
472        _, lab, times, is_trig = best
473        if clk is not None:
474            times = times + clk
475        out.append(Stream(
476            name=f"{name}::{lab}", times=times, both_edges=not is_trig,
477            kind="trigger events" if is_trig else "analog sync",
478            nominal_rate=float(info["nominal_srate"][0] or 0) or None,
479            n_samples=len(ts),
480            stream_start=float(ts[0]), stream_end=float(ts[-1]),
481            clock_offset=clk, sample_gaps=np.diff(ts)))
482        if verbose:
483            print(f"  {name}::{lab}  {len(times)} transitions")
484    return out

234def load_c3d_analog(filename: str) -> AnalogRecording:
235    """
236    Read analog channels from a C3D file, with no external C3D library.
237
238    Handles Intel/DEC/MIPS processor codes and both float and scaled-integer
239    storage. Marker data is skipped; this reader is for analog channels.
240
241    A .csv or .txt path is read as a Nexus ASCII export instead.
242    """
243    if not os.path.exists(filename):
244        raise FileNotFoundError(filename)
245
246    if filename.lower().endswith((".csv", ".txt")):
247        return _load_csv(filename)
248
249    with open(filename, "rb") as fh:
250        raw = fh.read()
251
252    if len(raw) < 512:
253        raise ValueError(f"{filename} is too short to be a C3D file.")
254
255    param_block = raw[0]
256    key = raw[1]
257    if key != 80:
258        raise ValueError(
259            f"{filename} is not a C3D file (header key {key}, expected 80).")
260
261    proc = _processor_type(raw, param_block)
262    endian = ">" if proc == 3 else "<"
263
264    def u16(off):
265        return struct.unpack_from(endian + "H", raw, off)[0]
266
267    def real(off):
268        if proc == 2:
269            return _dec_to_ieee(raw[off:off + 4])
270        return struct.unpack_from(endian + "f", raw, off)[0]
271
272    n_points         = u16(2)
273    analog_per_frame = u16(4)
274    first_frame      = u16(6)
275    last_frame       = u16(8)
276    point_scale      = real(12)
277    data_block       = u16(16)
278    analog_subframes = u16(18)
279    video_rate       = real(20)
280
281    if analog_subframes == 0 or analog_per_frame == 0:
282        raise ValueError(f"{filename} contains no analog channels.")
283
284    n_frames = last_frame - first_frame + 1
285    n_channels, rem = divmod(analog_per_frame, analog_subframes)
286    if rem:
287        raise ValueError("Analog channel count is not an integer.")
288
289    params = _read_parameters(raw, param_block, proc, endian)
290
291    fs = _param_scalar(params, "ANALOG", "RATE", video_rate * analog_subframes)
292    if not fs or fs <= 0:
293        fs = video_rate * analog_subframes
294
295    used = int(_param_scalar(params, "ANALOG", "USED", n_channels) or n_channels)
296    n_used = used if 0 < used <= n_channels else n_channels
297
298    labels = _param_strings(params, "ANALOG", "LABELS", n_channels)
299    units  = _param_strings(params, "ANALOG", "UNITS", n_channels)
300
301    scale  = _fit(_param_vector(params, "ANALOG", "SCALE", [1.0]), n_channels, 1.0)
302    offset = _fit(_param_vector(params, "ANALOG", "OFFSET", [0.0]), n_channels, 0.0)
303    gen    = _param_scalar(params, "ANALOG", "GEN_SCALE", 1.0) or 1.0
304
305    is_float = point_scale < 0
306    start = (data_block - 1) * 512
307    n_rows = n_frames * analog_subframes
308    n_total = n_rows * n_channels
309
310    if n_points == 0:
311        dtype = np.dtype(endian + ("f4" if is_float else "i2"))
312        need = n_total * dtype.itemsize
313        if start + need > len(raw):
314            raise ValueError(
315                f"File ended early: expected {need} bytes of analog data, "
316                f"found {len(raw) - start}.")
317        block = np.frombuffer(raw, dtype=dtype, count=n_total, offset=start)
318        values = block.reshape(n_rows, n_channels).astype(np.float64)
319    else:
320        # Point and analog data interleave frame by frame.
321        dtype = np.dtype(endian + ("f4" if is_float else "i2"))
322        per_frame_points = n_points * 4
323        per_frame_analog = analog_subframes * n_channels
324        stride = per_frame_points + per_frame_analog
325        need = n_frames * stride * dtype.itemsize
326        if start + need > len(raw):
327            raise ValueError("File ended early while reading interleaved data.")
328        block = np.frombuffer(raw, dtype=dtype,
329                              count=n_frames * stride, offset=start)
330        block = block.reshape(n_frames, stride)
331        values = block[:, per_frame_points:].reshape(n_rows, n_channels).astype(np.float64)
332
333    scale = np.asarray(scale, dtype=float)
334    offset = np.asarray(offset, dtype=float)
335    if is_float:
336        if np.any(scale != 1) or np.any(offset != 0):
337            values = (values - offset) * scale * gen
338    else:
339        values = (values - offset) * scale * gen

342        fs=float(fs),
343        labels=list(labels[:n_used]),
344        data=values[:, :n_used],
345        units=list(units[:n_used]),
346        filename=filename,
347    )

520                 fs: float,
521                 mode: str = "auto",
522                 hysteresis: Sequence[float] = (0.3, 0.7),
523                 threshold: float = 8.0,
524                 refractory: float = 30.0,
525                 min_pulse: float = 5.0,
526                 locate: str = "onset",
527                 onset_threshold: float = 2.0) -> Edges:
528    """
529    Find square-wave transition times to sub-sample precision.
530
531    Two acquisition paths need two detectors:
532
533    mode='level'
534        The square wave recorded directly. The signal holds its level between
535        transitions, so a Schmitt trigger with hysteresis finds the state
536        changes, and the crossing time is refined by linear interpolation
537        across the mid-level threshold.
538
539    mode='rectified'
540        The square wave after an EMG amplifier. The amplifier's high-pass
541        removes the DC level, so each step becomes a transient spike: a rising
542        edge gives a positive spike, a falling edge a negative one. The
543        detector therefore searches the SIGNED signal for positive and
544        negative peaks separately, so each detection carries a known polarity.
545        Peak times are refined by fitting a parabola through the peak and its
546        neighbours.
547

205    def _sliding(self, t):
206        """Local line fitted to the anchors bracketing each point.
207
208        Anchors are the decoded timecode frames. Rather than starting a
209        segment AT a frame — which leaves each boundary extrapolating in one
210        direction, exactly where the estimate is least supported — the
211        window SLIDES so the point sits between anchors on both sides.
212        Adjacent windows overlap, so a correction blends across a cut
213        instead of stepping at it.
214
215        Measured against a wandering clock (0 -> 400 ppm over 200 s), per
216        edge, versus ground truth:
217            one global line          median 5.21 ms   max 15.89 ms
218            segment starts at frame  median 0.68 ms   max  3.40 ms
219            sliding, centered        median 0.66 ms   max  3.28 ms
220        Widening the window to two anchors either side was worse (0.82 ms):
221        it over-smooths a clock that is genuinely changing.
222        """
223        ta, ga = self._anchor_t, self._anchor_g
224        n = len(ta)
225        out = np.empty(np.shape(t), dtype=float)
226        flat = np.atleast_1d(t).ravel()
227        res = np.empty(flat.size)
228        for i, xi in enumerate(flat):
229            k = int(np.searchsorted(ta, xi))
230            lo = max(0, min(k - 1, n - 4))
231            hi = min(n, lo + 4)
232            lo = max(0, hi - 4)
233            if hi - lo >= 2:
234                rate, off = np.polyfit(ta[lo:hi], ga[lo:hi], 1)
235            else:
236                rate, off = 1.0, ga[lo] - ta[lo]
237            res[i] = off + rate * xi

245        Times inside the dropped span map with the nearest segment, since no
246        samples actually exist there.
247        """
