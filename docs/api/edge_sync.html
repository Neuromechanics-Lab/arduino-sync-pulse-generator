edge_sync

edge_sync.py - Edge-based delay measurement and EMG processing.

Measures the delay between two or more recordings of the same square wave by locating the individual signal edges, rather than by cross-correlating the waveforms (which sync_align.py does). Edge timing is the better tool when one copy of the signal has been through an EMG amplifier, because the amplifier's high-pass turns each step into a transient spike: the waveforms no longer resemble each other, but their transition times still line up.

Companion to sync_align.py. Use that one for whole-waveform correlation, this one when you have clean edges and want a per-transition answer with error bars.

Usage as a module:

from edge_sync import load_c3d_analog, detect_edges, edge_delay, sync_report

rec = load_c3d_analog('SquareWaveTest01.c3d') rep = sync_report(rec) print(rep.delays_ms)

Usage as a CLI:

python edge_sync.py report SquareWaveTest01.c3d python edge_sync.py delay SquareWaveTest01.c3d --ref SquareDirect --test SquareWirelessEmg python edge_sync.py shift SquareWaveTest01.c3d --delay 22.99 -o corrected.csv

Dependencies: numpy. scipy is used for EMG filtering when available.

  48# Onset location: the level, as a fraction of each spike's own height, at
  49# which the rising flank is timed. Sitting partway up the flank rather than
  50# down in the noise makes the crossing insensitive to noise and to where the
  51# samples happen to fall; see _signed_peaks for why that matters.
  52#
  53# Choosing the value trades bias against spread, and both were measured
  54# against synthetic signals of known delay (1000 Hz, amplifier time constants
  55# 1-20 ms):
  56#
  57#   estimator      bias        varies with amplifier?
  58#   onset @ 0.25   +0.45 ms    no
  59#   onset @ 0.80   +1.00 ms    no
  60#   peak           +1.3..1.7   YES
  61#
  62# 0.25 gives the smallest bias, and - the property that matters most - a
  63# bias that does not depend on the amplifier's response. A fixed bias
  64# cancels out of any comparison between two channels recorded through the
  65# same hardware, whereas the peak's does not.
  66#
  67# The cost is noise: low on the flank the signal is still small, so on real
  68# recordings the spread is wider than the peak's, and raising the fraction
  69# trades bias back for precision. Report which you used.
  70ONSET_FRACTION = 0.25

  78class AnalogRecording:
  79    """Analog channels from a C3D or CSV export."""

  81    labels: list
  82    data: np.ndarray                  # samples x channels
  83    units: list = field(default_factory=list)

  87    def n_samples(self) -> int:

  94    @property
  95    def duration(self) -> float:
  96        return self.n_samples / self.fs
  97
  98    @property
  99    def time(self) -> np.ndarray:
 100        return np.arange(self.n_samples) / self.fs
 101
 102    @staticmethod
 103    def _bare(label: str) -> str:
 104        """Label without its device prefix: 'Voltage.5-TA' -> '5-TA'."""
 105        return label.rsplit(".", 1)[-1]
 106
 107    def find(self, pattern: str, unique: bool = False):
 108        """
 109        Channel indices whose label contains `pattern`, case-insensitive.
 110
 111        Matches against the label WITHOUT its device prefix first, falling
 112        back to the full label only if that finds nothing. The prefix would
 113        otherwise swallow short muscle abbreviations: Vicon writes channels
 114        as 'Voltage.5-TA', and a naive substring search for 'TA' hits the
 115        'ta' in every 'Voltage.', returning the entire device.
 116        """
 117        pat = pattern.lower()
 118        hits = [i for i, l in enumerate(self.labels)
 119                if pat in self._bare(l).lower()]
 120        if not hits:
 121            hits = [i for i, l in enumerate(self.labels) if pat in l.lower()]
 122        if unique:
 123            if not hits:
 124                raise ValueError(
 125                    f"No channel matching {pattern!r}. "
 126                    f"Available: {', '.join(self.labels)}")
 127            if len(hits) > 1:
 128                names = ", ".join(self.labels[i] for i in hits)
 129                raise ValueError(
 130                    f"Pattern {pattern!r} matched {len(hits)} channels: {names}")
 131            return hits[0]
 132        return hits
 133
 134    def channel(self, spec: Union[int, str]) -> np.ndarray:
 135        """One channel's samples, by index or name pattern."""
 136        idx = spec if isinstance(spec, (int, np.integer)) else self.find(spec, unique=True)
 137        return self.data[:, idx]

 233def load_c3d_analog(filename: str) -> AnalogRecording:
 234    """
 235    Read analog channels from a C3D file, with no external C3D library.
 236
 237    Handles Intel/DEC/MIPS processor codes and both float and scaled-integer
 238    storage. Marker data is skipped; this reader is for analog channels.
 239
 240    A .csv or .txt path is read as a Nexus ASCII export instead.
 241    """
 242    if not os.path.exists(filename):
 243        raise FileNotFoundError(filename)
 244
 245    if filename.lower().endswith((".csv", ".txt")):
 246        return _load_csv(filename)
 247
 248    with open(filename, "rb") as fh:
 249        raw = fh.read()
 250
 251    if len(raw) < 512:
 252        raise ValueError(f"{filename} is too short to be a C3D file.")
 253
 254    param_block = raw[0]
 255    key = raw[1]
 256    if key != 80:
 257        raise ValueError(
 258            f"{filename} is not a C3D file (header key {key}, expected 80).")
 259
 260    proc = _processor_type(raw, param_block)
 261    endian = ">" if proc == 3 else "<"
 262
 263    def u16(off):
 264        return struct.unpack_from(endian + "H", raw, off)[0]
 265
 266    def real(off):
 267        if proc == 2:
 268            return _dec_to_ieee(raw[off:off + 4])
 269        return struct.unpack_from(endian + "f", raw, off)[0]
 270
 271    n_points         = u16(2)
 272    analog_per_frame = u16(4)
 273    first_frame      = u16(6)
 274    last_frame       = u16(8)
 275    point_scale      = real(12)
 276    data_block       = u16(16)
 277    analog_subframes = u16(18)
 278    video_rate       = real(20)
 279
 280    if analog_subframes == 0 or analog_per_frame == 0:
 281        raise ValueError(f"{filename} contains no analog channels.")
 282
 283    n_frames = last_frame - first_frame + 1
 284    n_channels, rem = divmod(analog_per_frame, analog_subframes)
 285    if rem:
 286        raise ValueError("Analog channel count is not an integer.")
 287
 288    params = _read_parameters(raw, param_block, proc, endian)
 289
 290    fs = _param_scalar(params, "ANALOG", "RATE", video_rate * analog_subframes)
 291    if not fs or fs <= 0:
 292        fs = video_rate * analog_subframes
 293
 294    used = int(_param_scalar(params, "ANALOG", "USED", n_channels) or n_channels)
 295    n_used = used if 0 < used <= n_channels else n_channels
 296
 297    labels = _param_strings(params, "ANALOG", "LABELS", n_channels)
 298    units  = _param_strings(params, "ANALOG", "UNITS", n_channels)
 299
 300    scale  = _fit(_param_vector(params, "ANALOG", "SCALE", [1.0]), n_channels, 1.0)
 301    offset = _fit(_param_vector(params, "ANALOG", "OFFSET", [0.0]), n_channels, 0.0)
 302    gen    = _param_scalar(params, "ANALOG", "GEN_SCALE", 1.0) or 1.0
 303
 304    is_float = point_scale < 0
 305    start = (data_block - 1) * 512
 306    n_rows = n_frames * analog_subframes
 307    n_total = n_rows * n_channels
 308
 309    if n_points == 0:
 310        dtype = np.dtype(endian + ("f4" if is_float else "i2"))
 311        need = n_total * dtype.itemsize
 312        if start + need > len(raw):
 313            raise ValueError(
 314                f"File ended early: expected {need} bytes of analog data, "
 315                f"found {len(raw) - start}.")
 316        block = np.frombuffer(raw, dtype=dtype, count=n_total, offset=start)
 317        values = block.reshape(n_rows, n_channels).astype(np.float64)
 318    else:
 319        # Point and analog data interleave frame by frame.
 320        dtype = np.dtype(endian + ("f4" if is_float else "i2"))
 321        per_frame_points = n_points * 4
 322        per_frame_analog = analog_subframes * n_channels
 323        stride = per_frame_points + per_frame_analog
 324        need = n_frames * stride * dtype.itemsize
 325        if start + need > len(raw):
 326            raise ValueError("File ended early while reading interleaved data.")
 327        block = np.frombuffer(raw, dtype=dtype,
 328                              count=n_frames * stride, offset=start)
 329        block = block.reshape(n_frames, stride)
 330        values = block[:, per_frame_points:].reshape(n_rows, n_channels).astype(np.float64)
 331
 332    scale = np.asarray(scale, dtype=float)
 333    offset = np.asarray(offset, dtype=float)
 334    if is_float:
 335        if np.any(scale != 1) or np.any(offset != 0):
 336            values = (values - offset) * scale * gen
 337    else:
 338        values = (values - offset) * scale * gen
 339
 340    return AnalogRecording(
 341        fs=float(fs),
 342        labels=list(labels[:n_used]),
 343        data=values[:, :n_used],
 344        units=list(units[:n_used]),
 345        filename=filename,
 346    )

 349def _processor_type(raw: bytes, param_block: int) -> int:
 350    off = (param_block - 1) * 512 + 3
 351    if off + 1 >= len(raw):
 352        return 1
 353    code = raw[off + 1]
 354    return {85: 2, 86: 3}.get(code, 1)

 357def _dec_to_ieee(b: bytes) -> float:
 358    if len(b) < 4 or not any(b):
 359        return 0.0
 360    swapped = bytes([b[2], b[3], b[0], b[1]])
 361    word = struct.unpack("<I", swapped)[0]
 362    sign = (word >> 31) & 1
 363    expo = (word >> 23) & 0xFF
 364    mant = word & 0x7FFFFF
 365    if expo == 0:
 366        return 0.0
 367    val = (1 + mant / 2**23) * 2.0 ** (expo - 129)
 368    return -val if sign else val

 371def _read_parameters(raw: bytes, param_block: int, proc: int, endian: str) -> dict:
 372    """Walk the parameter linked list into {GROUP: {PARAM: (type, dims, data)}}."""
 373    params: dict = {}
 374    groups: dict = {}
 375    base = (param_block - 1) * 512
 376    off = base + 4
 377    guard = 0
 378
 379    while off + 2 <= len(raw):
 380        guard += 1
 381        if guard > 20000:
 382            break
 383        n_char = struct.unpack_from("b", raw, off)[0]
 384        if n_char == 0:
 385            break
 386        gid = struct.unpack_from("b", raw, off + 1)[0]
 387        name_end = off + 2 + abs(n_char)
 388        if name_end + 2 > len(raw):
 389            break
 390        name = raw[off + 2:name_end].decode("ascii", "replace").strip().upper()
 391        next_off = struct.unpack_from(endian + "h", raw, name_end)[0]
 392        cursor = name_end + 2
 393
 394        if gid < 0:
 395            groups[abs(gid)] = name
 396        else:
 397            if cursor + 2 > len(raw):
 398                break
 399            dtype_code = struct.unpack_from("b", raw, cursor)[0]
 400            n_dims = raw[cursor + 1]
 401            dims = list(raw[cursor + 2:cursor + 2 + n_dims]) or [1]
 402            dstart = cursor + 2 + n_dims
 403            n_elem = int(np.prod(dims))
 404
 405            if dtype_code == -1:
 406                data = raw[dstart:dstart + n_elem].decode("ascii", "replace")
 407            elif dtype_code == 1:
 408                data = list(struct.unpack_from(f"{endian}{n_elem}b", raw, dstart))
 409            elif dtype_code == 2:
 410                data = list(struct.unpack_from(f"{endian}{n_elem}h", raw, dstart))
 411            elif dtype_code == 4:
 412                if proc == 2:
 413                    data = [_dec_to_ieee(raw[dstart + 4*k:dstart + 4*k + 4])
 414                            for k in range(n_elem)]
 415                else:
 416                    data = list(struct.unpack_from(f"{endian}{n_elem}f", raw, dstart))
 417            else:
 418                data = []
 419
 420            gname = groups.get(gid, f"GROUP{gid}")
 421            params.setdefault(gname, {})[name] = (dtype_code, dims, data)
 422
 423        if next_off == 0:
 424            break
 425        new_off = name_end + next_off
 426        if new_off <= off:
 427            break
 428        off = new_off
 429
 430    return params

 447def _param_strings(params, group, name, n_expected):
 448    out = [f"Channel{i+1}" for i in range(n_expected)]
 449    rec = params.get(group, {}).get(name)
 450    if not rec or rec[0] != -1 or len(rec[1]) < 2:
 451        return out
 452    width, count = rec[1][0], rec[1][1]
 453    chars = rec[2]
 454    for i in range(min(count, n_expected)):
 455        s = chars[i*width:(i+1)*width].strip()
 456        if s:
 457            out[i] = s
 458    return out

 461def _fit(vals, n, fill):
 462    vals = list(vals)
 463    if len(vals) == 1:
 464        return vals * n
 465    if len(vals) < n:
 466        return vals + [fill] * (n - len(vals))
 467    return vals[:n]

 470def _load_csv(filename: str) -> AnalogRecording:
 471    """Read a Nexus ASCII export, or a plain CSV with a time column."""
 472    with open(filename, "r") as fh:
 473        head = [fh.readline().rstrip("\n") for _ in range(200)]
 474
 475    dev_idx = next((i for i, l in enumerate(head)
 476                    if l.strip().lower() == "devices"), None)
 477
 478    if dev_idx is not None and dev_idx + 3 < len(head):
 479        fs = float(head[dev_idx + 1].split(",")[0])
 480        labels = [c.strip() for c in head[dev_idx + 3].split(",")]
 481        arr = np.genfromtxt(filename, delimiter=",", skip_header=dev_idx + 4)
 482        data = arr[:, 2:]
 483        labels = labels[2:2 + data.shape[1]]
 484    else:
 485        arr = np.genfromtxt(filename, delimiter=",", names=True)
 486        labels = list(arr.dtype.names)
 487        data = np.column_stack([arr[n] for n in labels])
 488        fs = None
 489        for cand in ("time", "t", "Time"):
 490            if cand in labels:
 491                col = labels.index(cand)
 492                fs = 1.0 / float(np.median(np.diff(data[:, col])))
 493                data = np.delete(data, col, axis=1)
 494                labels.pop(col)
 495                break
 496        if fs is None:
 497            raise ValueError(
 498                f"Could not determine the sample rate from {filename}. "
 499                "Export with the Nexus rate header, or use a .c3d file.")
 500
 501    while len(labels) < data.shape[1]:
 502        labels.append(f"Channel{len(labels)+1}")
 503
 504    return AnalogRecording(fs=float(fs), labels=labels[:data.shape[1]],
 505                           data=data, units=[""] * data.shape[1],
 506                           filename=filename)

 518def detect_edges(signal: np.ndarray,
 519                 fs: float,
 520                 mode: str = "auto",
 521                 hysteresis: Sequence[float] = (0.3, 0.7),
 522                 threshold: float = 8.0,
 523                 refractory: float = 30.0,
 524                 min_pulse: float = 5.0,
 525                 locate: str = "onset",
 526                 onset_threshold: float = 2.0) -> Edges:
 527    """
 528    Find square-wave transition times to sub-sample precision.
 529
 530    Two acquisition paths need two detectors:
 531
 532    mode='level'
 533        The square wave recorded directly. The signal holds its level between
 534        transitions, so a Schmitt trigger with hysteresis finds the state
 535        changes, and the crossing time is refined by linear interpolation
 536        across the mid-level threshold.
 537
 538    mode='rectified'
 539        The square wave after an EMG amplifier. The amplifier's high-pass
 540        removes the DC level, so each step becomes a transient spike: a rising
 541        edge gives a positive spike, a falling edge a negative one. The
 542        detector therefore searches the SIGNED signal for positive and
 543        negative peaks separately, so each detection carries a known polarity.
 544        Peak times are refined by fitting a parabola through the peak and its
 545        neighbours.
 546
 547        Working on the signed signal rather than a rectified copy is the whole
 548        point: rectifying discards the sign, and without the sign a rising
 549        edge can be paired with a falling one. See edge_delay.
 550
 551    mode='auto'
 552        Chooses by looking at how much time the signal spends near mid-range.
 553
 554    Parameters
 555    ----------
 556    hysteresis : Schmitt thresholds as fractions of the signal range.
 557    threshold  : rectified mode, peak threshold in MAD units.
 558    refractory : rectified mode, minimum ms between peaks of one polarity.
 559                 Stops post-transition ringing counting as a second edge.
 560                 Must be shorter than the shortest pulse in the pattern.
 561    min_pulse  : level mode, minimum ms a level must hold.
 562    locate     : rectified mode, which feature of the spike marks the
 563                 transition. 'onset' (default) times the rising flank;
 564                 'peak' uses the spike maximum.
 565
 566                 Both are biased late; what separates them is whether the
 567                 bias is CONSTANT. The PEAK lags by the amplifier's rise
 568                 time, so its bias grows with the amplifier's time constant
 569                 (+1.3 to +1.7 ms across 1-20 ms in testing). The ONSET is
 570                 timed at a fixed fraction of each spike's height, so its
 571                 bias is the same whatever the amplifier (+0.45 ms at the
 572                 default fraction).
 573
 574                 A constant bias cancels out of a comparison between two
 575                 channels recorded through the same hardware; a varying one
 576                 does not. Onset is therefore the default. Use 'peak' to
 577                 reproduce an older analysis, or when the flank is too noisy
 578                 for a stable onset.
 579    onset_threshold : absolute floor for the onset level, in MAD units, so a
 580                 tiny spike cannot be timed inside the noise. The level
 581                 actually used is the larger of this and ONSET_FRACTION of
 582                 the spike's own height.
 583    """
 584    x = np.asarray(signal, dtype=float).ravel()
 585    if x.size < 3:
 586        raise ValueError("Signal needs at least 3 samples.")
 587    if fs <= 0:
 588        raise ValueError("fs must be positive.")
 589
 590    noise = _mad(x)
 591
 592    if mode == "auto":
 593        mode = _classify(x)
 594
 595    if mode == "level":
 596        t, p, a = _level_edges(x, fs, hysteresis, min_pulse)
 597    elif mode == "rectified":
 598        t, p, a = _rectified_edges(x, fs, noise, threshold, refractory,
 599                                   locate, onset_threshold)
 600    else:
 601        raise ValueError("mode must be 'level', 'rectified', or 'auto'.")
 602
 603    return Edges(time=t, polarity=p, amplitude=a, mode=mode, fs=fs, noise=noise)

 606def _classify(x: np.ndarray) -> str:
 607    """A directly recorded square wave is bimodal; a differentiated one is not."""
 608    lo, hi = float(np.min(x)), float(np.max(x))
 609    rng = hi - lo
 610    if rng <= 0:
 611        return "level"
 612    mid_frac = np.mean((x > lo + 0.35 * rng) & (x < lo + 0.65 * rng))
 613    return "level" if mid_frac < 0.10 else "rectified"

 616def _level_edges(x, fs, hysteresis, min_pulse):
 617    lo, hi = float(np.min(x)), float(np.max(x))
 618    rng = hi - lo
 619    if rng <= 0:
 620        return np.array([]), np.array([]), np.array([])
 621
 622    f_lo, f_hi = sorted(hysteresis)
 623    thr_lo = lo + f_lo * rng
 624    thr_hi = lo + f_hi * rng
 625    mid = (hi + lo) / 2.0
 626    min_gap = max(1, int(round(min_pulse * fs / 1000.0)))
 627
 628    times, pols = [], []
 629    state = x[0] > thr_hi
 630    last_idx = -np.inf
 631
 632    for i in range(1, x.size):
 633        up = (not state) and x[i] > thr_hi
 634        down = state and x[i] < thr_lo
 635        if not (up or down):
 636            continue
 637        state = up
 638
 639        j = i
 640        while j > 0 and ((x[j] > mid) if up else (x[j] < mid)):
 641            j -= 1
 642
 643        if j < x.size - 1:
 644            y0, y1 = x[j], x[j + 1]
 645            frac = (mid - y0) / (y1 - y0) if y1 != y0 else 0.0
 646            frac = min(max(frac, 0.0), 1.0)
 647            t_cross = (j + frac) / fs
 648        else:
 649            t_cross = j / fs
 650
 651        if (j - last_idx) < min_gap:
 652            continue
 653        last_idx = j
 654
 655        times.append(t_cross)
 656        pols.append(1 if up else -1)
 657
 658    times = np.asarray(times, dtype=float)
 659    pols = np.asarray(pols, dtype=int)
 660    return times, pols, np.full(times.shape, rng)

 663def _rectified_edges(x, fs, noise, threshold, refractory,
 664                     locate="onset", onset_threshold=2.0):
 665    if noise <= 0:
 666        return np.array([]), np.array([]), np.array([])
 667
 668    xc = x - np.median(x)
 669    thr = threshold * noise
 670
 671    tp, ap = _signed_peaks(xc, +1, thr, fs, refractory,
 672                           locate, onset_threshold * noise)
 673    tn, an = _signed_peaks(xc, -1, thr, fs, refractory,
 674                           locate, onset_threshold * noise)
 675
 676    t = np.concatenate([tp, tn])
 677    p = np.concatenate([np.ones(tp.size, dtype=int), -np.ones(tn.size, dtype=int)])
 678    a = np.concatenate([ap, an])
 679
 680    order = np.argsort(t)
 681    return t[order], p[order], a[order]

 684def _signed_peaks(x, sign, thr, fs, refractory_ms,
 685                  locate="onset", onset_level=None):
 686    """
 687    Peaks of one polarity, held apart by a refractory period.
 688
 689    When two candidates fall inside the refractory window the LARGER wins, so
 690    ringing after a real transition cannot displace the transition itself.
 691
 692    `locate` selects which feature of each spike is reported as the edge
 693    time: 'peak' for the maximum, 'onset' for a fixed fraction of the way up
 694    the rising flank. Both lag the true transition, but the onset's lag does
 695    not depend on the amplifier's response, so it cancels out of a delay
 696    measurement between two channels; the peak's does not.
 697    """
 698    s = x * sign
 699    n = s.size
 700    above = s > thr
 701    if not above.any():
 702        return np.array([]), np.array([])
 703
 704    # One candidate per supra-threshold run.
 705    edges_ = np.diff(above.astype(np.int8))
 706    starts = np.flatnonzero(edges_ == 1) + 1
 707    stops = np.flatnonzero(edges_ == -1) + 1
 708    if above[0]:
 709        starts = np.r_[0, starts]
 710    if above[-1]:
 711        stops = np.r_[stops, n]
 712
 713    cand_idx, cand_val = [], []
 714    for a, b in zip(starts, stops):
 715        k = a + int(np.argmax(s[a:b]))
 716        cand_idx.append(k)
 717        cand_val.append(s[k])
 718
 719    cand_idx = np.asarray(cand_idx)
 720    cand_val = np.asarray(cand_val)
 721
 722    # Greedy largest-first under the refractory constraint.
 723    refrac = max(1, int(round(refractory_ms * fs / 1000.0)))
 724    kept = []
 725    for q in np.argsort(-cand_val):
 726        ci = cand_idx[q]
 727        if all(abs(ci - k) >= refrac for k in kept):
 728            kept.append(ci)
 729    kept = np.sort(np.asarray(kept, dtype=int))
 730
 731    times = np.empty(kept.size)
 732    amps = np.empty(kept.size)
 733    if onset_level is None:
 734        onset_level = thr / 4.0
 735
 736    for m, k in enumerate(kept):
 737        amps[m] = s[k]
 738
 739        if locate == "onset":
 740            # Walk back from the peak to where the rising flank crosses a
 741            # level set as a FRACTION OF THIS SPIKE'S OWN AMPLITUDE, then
 742            # interpolate that crossing.
 743            #
 744            # An absolute threshold (a few MADs) sits in the noise at the very
 745            # foot of the flank, where the signal is nearly flat: which sample
 746            # first exceeds it then depends on noise, and the interpolated
 747            # time snaps to one sample or the next. On real data that split a
 748            # single population into two modes ~1 sample apart. A fractional
 749            # level sits on the steep part of the flank, where a small
 750            # amplitude error moves the crossing time very little, and it
 751            # scales automatically with spike size.
 752            #
 753            # The level is still a fixed fraction of the rise, so it remains a
 754            # constant delay after the true transition rather than tracking
 755            # it exactly - but that delay is identical for every edge of a
 756            # given shape, so it cancels out of a delay measurement.
 757            level = max(onset_level, ONSET_FRACTION * s[k])
 758            j = k
 759            while j > 0 and s[j] > level:
 760                j -= 1
 761            if j < k:
 762                y0, y1 = s[j], s[j + 1]
 763                frac = (level - y0) / (y1 - y0) if y1 != y0 else 0.0
 764                frac = min(max(frac, 0.0), 1.0)
 765                times[m] = (j + frac) / fs
 766            else:
 767                times[m] = j / fs
 768            continue
 769
 770        # locate == 'peak': parabolic interpolation through the maximum.
 771        if 0 < k < n - 1:
 772            y0, y1, y2 = s[k - 1], s[k], s[k + 1]
 773            den = y0 - 2 * y1 + y2
 774            delta = 0.5 * (y0 - y2) / den if den != 0 else 0.0
 775            delta = min(max(delta, -1.0), 1.0)
 776        else:
 777            delta = 0.0
 778        times[m] = (k + delta) / fs
 779
 780    return times, amps

 787def edge_delay(edges_ref: Edges,
 788               edges_test: Edges,
 789               min_delay: float = -2.0,
 790               max_delay: float = 100.0,
 791               outlier_mad: float = 5.0) -> DelayResult:
 792    """
 793    Delay between two recordings of the same square wave, from their edges.
 794
 795    A positive delay means the test signal arrives AFTER the reference.
 796
 797    Matching rules, and why each one is there:
 798
 799    1. Polarity is respected. A rising edge is only ever matched to a rising
 800       edge. Mixing polarities inflates the spread, because a rising and a
 801       falling transition are different events separated by a pulse width.
 802
 803    2. Matching is causal. A candidate must fall within
 804       [min_delay, max_delay] of the reference edge. The test signal cannot
 805       physically precede the reference, so a negative-delay pairing is a
 806       detection artefact and is excluded rather than averaged in.
 807
 808    3. Ties go to the largest peak. When several candidates sit inside the
 809       window, the strongest wins. Spurious peaks are small; the real
 810       transition is not.
 811
 812    Together these turn a nearest-neighbour match into one that will not
 813    silently pair a real edge with noise.
 814    """
 815    if min_delay >= max_delay:
 816        raise ValueError("min_delay must be below max_delay.")
 817    if len(edges_ref) == 0 or len(edges_test) == 0:
 818        raise ValueError(
 819            f"Both inputs need edges (reference {len(edges_ref)}, "
 820            f"test {len(edges_test)}).")
 821
 822    amp = np.abs(edges_test.amplitude) if edges_test.amplitude.size \
 823        else np.ones(len(edges_test))
 824
 825    tr, dr = _match(edges_ref, edges_test, amp, +1, min_delay, max_delay)
 826    tf, df = _match(edges_ref, edges_test, amp, -1, min_delay, max_delay)
 827
 828    times = np.concatenate([tr, tf])
 829    deltas = np.concatenate([dr, df])
 830    pols = np.concatenate([np.ones(tr.size, dtype=int),
 831                           -np.ones(tf.size, dtype=int)])
 832
 833    if deltas.size == 0:
 834        raise ValueError(
 835            f"No edges paired inside [{min_delay}, {max_delay}] ms. Check that "
 836            "both channels carry the same square wave, and widen max_delay if "
 837            "the true delay could exceed it.")
 838
 839    order = np.argsort(times)
 840    times, deltas, pols = times[order], deltas[order], pols[order]
 841
 842    if deltas.size >= 3 and np.ptp(times) > 0:
 843        slope, intercept = np.polyfit(times, deltas, 1)
 844        drift_total = slope * np.ptp(times)
 845    else:
 846        slope, intercept, drift_total = 0.0, float(np.median(deltas)), 0.0
 847
 848    med = float(np.median(deltas))
 849    mad = 1.4826 * np.median(np.abs(deltas - med))
 850    if outlier_mad > 0 and mad > 0:
 851        bad = np.abs(deltas - med) > outlier_mad * mad
 852    else:
 853        bad = np.zeros(deltas.shape, dtype=bool)
 854
 855    q75, q25 = np.percentile(deltas, [75, 25])
 856
 857    result = DelayResult(
 858        delay_ms=med,
 859        delay_mean_ms=float(np.mean(deltas)),
 860        delay_std_ms=float(np.std(deltas, ddof=1)) if deltas.size > 1 else 0.0,
 861        delay_iqr_ms=float(q75 - q25),
 862        ci95_ms=float(1.96 * np.std(deltas, ddof=1) / np.sqrt(deltas.size))
 863        if deltas.size > 1 else 0.0,
 864        n_matched=int(deltas.size),
 865        n_reference=len(edges_ref),
 866        match_rate=float(deltas.size / len(edges_ref)),
 867        rising=_stats(dr),
 868        falling=_stats(df),
 869        drift_ms_per_s=float(slope),
 870        drift_total_ms=float(drift_total),
 871        intercept_ms=float(intercept),
 872        times=times,
 873        deltas_ms=deltas,
 874        polarities=pols,
 875        n_outliers=int(bad.sum()),
 876        outlier_idx=np.flatnonzero(bad),
 877        warnings=[],
 878    )
 879    result.warnings = _quality(result, edges_ref, edges_test)
 880    return result

 883def _match(e_ref, e_test, amp, pol, min_delay, max_delay):
 884    """Pair each reference edge with the strongest same-polarity test edge
 885    inside the causal window."""
 886    ref_t = e_ref.time[e_ref.polarity == pol]
 887    sel = e_test.polarity == pol
 888    tst_t = e_test.time[sel]
 889    tst_a = amp[sel]
 890
 891    if tst_t.size == 0 or ref_t.size == 0:
 892        return np.array([]), np.array([])
 893
 894    times, deltas = [], []
 895    for t0 in ref_t:
 896        d_ms = (tst_t - t0) * 1000.0
 897        inwin = (d_ms >= min_delay) & (d_ms <= max_delay)
 898        if not inwin.any():
 899            continue
 900        idx = np.flatnonzero(inwin)
 901        best = idx[int(np.argmax(tst_a[idx]))]
 902        times.append(t0)
 903        deltas.append(d_ms[best])
 904
 905    return np.asarray(times), np.asarray(deltas)

 908def _stats(d):
 909    if d.size == 0:
 910        return {"median_ms": float("nan"), "mean_ms": float("nan"),
 911                "std_ms": float("nan"), "n": 0}
 912    return {
 913        "median_ms": float(np.median(d)),
 914        "mean_ms": float(np.mean(d)),
 915        "std_ms": float(np.std(d, ddof=1)) if d.size > 1 else 0.0,
 916        "n": int(d.size),
 917    }

 920def _quality(r: DelayResult, e_ref: Edges, e_test: Edges) -> list:
 921    """Turn the numbers into plain statements about whether to trust them."""
 922    w = []
 923    if r.match_rate < 0.9:
 924        w.append(f"Only {100*r.match_rate:.0f}% of reference edges matched "
 925                 f"({r.n_matched} of {r.n_reference}). Check the detector "
 926                 f"settings on the test channel.")
 927    if r.n_outliers:
 928        w.append(f"{r.n_outliers} matched edges are more than 5 MADs from the "
 929                 f"median. Inspect them before trusting the mean.")
 930    if r.delay_std_ms > 2:
 931        w.append(f"Delay spread is {r.delay_std_ms:.2f} ms (sd). A fixed "
 932                 f"hardware latency should be well under 1 ms; this suggests "
 933                 f"detection problems or a genuinely variable link.")
 934    rm, fm = r.rising["median_ms"], r.falling["median_ms"]
 935    if np.isfinite(rm) and np.isfinite(fm) and abs(rm - fm) > 1:
 936        w.append(f"Rising and falling edges disagree by {abs(rm-fm):.2f} ms "
 937                 f"({rm:.3f} vs {fm:.3f}). Asymmetry this large usually means "
 938                 f"the detector is mis-locating one polarity.")
 939    if abs(r.drift_total_ms) > 1:
 940        w.append(f"Delay drifts {r.drift_total_ms:.2f} ms across the trial. The "
 941                 f"two devices may run on independent clocks; a single offset "
 942                 f"will not align them properly.")
 943    # Only compare edge counts when BOTH channels used the same detector.
 944    #
 945    # A level detector reports one edge per transition. A rectified detector
 946    # reports a positive AND a negative peak at every transition, because the
 947    # amplifier's response overshoots on the way back: a step up gives a
 948    # positive spike followed by a negative one. So a correctly detected
 949    # rectified channel legitimately has about twice the reference's edges
 950    # in EVERY polarity, and flagging that ratio calls a good detection bad.
 951    #
 952    # The real check on whether the pairing worked is the match rate and the
 953    # rising/falling agreement, both already covered above.
 954    if e_ref.mode == e_test.mode and len(e_ref) and len(e_test) > 1.5 * len(e_ref):
 955        w.append(f"Test channel yielded {len(e_test)} edges against "
 956                 f"{len(e_ref)} in the reference, using the same detector. "
 957                 f"Raise 'threshold' or 'refractory' if spurious peaks are "
 958                 f"being detected.")
 959    return w

 966def sync_report(rec: AnalogRecording,
 967                channels: Optional[Sequence] = None,
 968                reference: Optional[Union[int, str]] = None,
 969                max_delay: float = 100.0) -> SyncReport:
 970    """
 971    Measure the delay of every sync channel against a reference.
 972
 973    The one-call entry point for: given a recording carrying the same square
 974    wave on two or more channels, how far apart are they? Handles any number
 975    of sync channels, not only two.
 976
 977    With `channels` unset it auto-detects any channel whose label mentions
 978    'square', 'sync', or 'trig' and which actually contains transitions.
 979    Every non-reference channel is compared against the reference, so N sync
 980    channels give N-1 delays on a common baseline.
 981    """
 982    if channels is None:
 983        idx = _autodetect(rec)
 984        if len(idx) < 2:
 985            raise ValueError(
 986                f"Found {len(idx)} sync channel(s) automatically. Name them "
 987                f"explicitly with `channels`. Labels present: "
 988                f"{', '.join(rec.labels)}")
 989    else:
 990        idx = [c if isinstance(c, (int, np.integer)) else rec.find(c, unique=True)
 991               for c in channels]
 992
 993    if reference is None:
 994        ref_ch = idx[0]
 995    else:
 996        ref_ch = reference if isinstance(reference, (int, np.integer)) \
 997            else rec.find(reference, unique=True)
 998        if ref_ch not in idx:
 999            idx = [ref_ch] + list(idx)
1000
1001    others = [i for i in idx if i != ref_ch]
1002    all_idx = [ref_ch] + others
1003
1004    edges = [detect_edges(rec.data[:, i], rec.fs) for i in all_idx]
1005
1006    results, delays, warns = [], [], []
1007    for k, ch in enumerate(others):
1008        r = edge_delay(edges[0], edges[k + 1], max_delay=max_delay)
1009        results.append(r)
1010        delays.append(r.delay_ms)
1011        warns += [f"[{rec.labels[ch]}] {w}" for w in r.warnings]
1012
1013    n = len(all_idx)
1014    M = np.full((n, n), np.nan)
1015    for a in range(n):
1016        M[a, a] = 0.0
1017        for b in range(a + 1, n):
1018            try:
1019                rab = edge_delay(edges[a], edges[b], max_delay=max_delay)
1020                M[a, b] = rab.delay_ms
1021                M[b, a] = -rab.delay_ms
1022            except ValueError:
1023                pass
1024
1025    return SyncReport(
1026        reference=rec.labels[ref_ch],
1027        ref_index=ref_ch,
1028        channels=[rec.labels[i] for i in others],
1029        ch_indices=others,
1030        all_indices=all_idx,
1031        delays_ms=np.asarray(delays),
1032        results=results,
1033        edges=edges,
1034        pairwise_ms=M,
1035        pairwise_labels=[rec.labels[i] for i in all_idx],
1036        warnings=warns,
1037    )

1040def _autodetect(rec: AnalogRecording) -> list:
1041    """Channels whose label mentions a sync signal AND which carry edges."""
1042    idx = [i for i, l in enumerate(rec.labels)
1043           if any(k in l.lower() for k in ("square", "sync", "trig"))]
1044    keep = []
1045    for i in idx:
1046        x = rec.data[:, i]
1047        if np.std(x) > 0 and len(detect_edges(x, rec.fs)) >= 4:
1048            keep.append(i)
1049    return keep

1056def shift_timestamps(rec: AnalogRecording,
1057                     delay_ms: Union[float, DelayResult],
1058                     channels: Optional[Sequence[int]] = None,
1059                     resample: bool = False,
1060                     invert: bool = False,
1061                     method: str = "linear"):
1062    """
1063    Correct a recording's timestamps by a measured delay.
1064
1065    `delay_ms` may be a number or a DelayResult, whose .delay_ms is used.
1066
1067    Sign convention matches edge_delay: a positive delay means this recording
1068    arrived LATE, so its timestamps move EARLIER (the sample labelled t really
1069    happened at t - delay). Pass invert=True to reverse that.
1070
1071    Two styles:
1072
1073    Timestamp shift (default)
1074        Rewrites the time vector, leaves the samples untouched. Nothing is
1075        interpolated, so no data is altered.
1076
1077    Resample (resample=True)
1078        Keeps the original time base and moves the DATA onto it by
1079        interpolation. Use when something downstream insists every recording
1080        share one grid. Interpolation is lossy, hence not the default.
1081
1082    Returns (time, data, info) where info records what was applied.
1083    """
1084    d_ms = delay_ms.delay_ms if isinstance(delay_ms, DelayResult) else float(delay_ms)
1085    if not np.isfinite(d_ms):
1086        raise ValueError("delay_ms must be finite.")
1087    if invert:
1088        d_ms = -d_ms
1089    d_s = d_ms / 1000.0
1090
1091    t = rec.time
1092    data = rec.data
1093    ch = list(range(rec.n_channels)) if channels is None else list(channels)
1094    if any(c < 0 or c >= rec.n_channels for c in ch):
1095        raise ValueError(f"Channel indices must be in [0, {rec.n_channels-1}].")
1096
1097    info = {"shift_applied_ms": d_ms, "shift_channels": ch,
1098            "resampled": resample, "original_time": t}
1099
1100    if not resample:
1101        if len(ch) == rec.n_channels:
1102            return t - d_s, data, info
1103        info["shifted_time"] = t - d_s
1104        info["note"] = (f"Only {len(ch)} of {rec.n_channels} channels were "
1105                        f"shifted; their time base is in info['shifted_time'].")
1106        return t, data, info
1107
1108    src = t - d_s
1109    out = data.copy()
1110    for c in ch:
1111        v = data[:, c].astype(float)
1112        good = np.isfinite(v)
1113        if not good.any():
1114            continue
1115        out[:, c] = np.interp(t, src[good], v[good], left=np.nan, right=np.nan)
1116    return t, out, info

1125                band: Optional[Sequence[float]] = (20.0, 450.0),

1127                notch: Optional[float] = None,
1128                notch_q: float = 30.0,
1129                envelope: Optional[float] = 4.0,
1130                rms_window: float = 100.0,
1131                mvc: Optional[float] = None) -> dict:
1132    """
1133    Standard surface-EMG processing chain, returning every intermediate stage.
1134
1135        raw -> detrend -> bandpass -> (notch) -> rectify -> envelope
1136
1137    The bandpass is a zero-phase Butterworth applied with filtfilt, so it adds
1138    no phase lag - which matters here, since a lag would corrupt the very
1139    timing this module measures.
1140
1141    The returned dict also carries a 'quality' block and plain-language
1142    'warnings', which flag flat channels (nothing plugged in) and railed ones
1143    (a disconnected sensor floats to the supply limits and can look like a
1144    large signal unless checked).
1145    """
1146    x = np.asarray(signal, dtype=float).ravel()
1147    if x.size < 10:
1148        raise ValueError("Signal needs at least 10 samples.")
1149    nyq = fs / 2.0
1150
1151    out = {"raw": x, "fs": fs, "time": np.arange(x.size) / fs}
1152    out["quality"] = _emg_quality(x, fs, notch)
1153
1154    y = x - np.mean(x)
1155    out["detrended"] = y
1156
1157    if band is not None and len(band) == 2:
1158        lo, hi = float(band[0]), min(float(band[1]), 0.99 * nyq)
1159        if lo >= hi:
1160            raise ValueError(f"Bandpass {band} is invalid at fs={fs} Hz.")
1161        if not HAVE_SCIPY:
1162            raise ImportError("scipy is required for EMG filtering.")
1163        b, a = sp_signal.butter(order, [lo / nyq, hi / nyq], btype="bandpass")
1164        y = sp_signal.filtfilt(b, a, y)
1165    out["filtered"] = y
1166
1167    if notch and 0 < notch < nyq:
1168        if not HAVE_SCIPY:
1169            raise ImportError("scipy is required for the notch filter.")
1170        b, a = sp_signal.iirnotch(notch / nyq, notch_q)
1171        y = sp_signal.filtfilt(b, a, y)
1172        out["notched"] = y
1173        out["filtered"] = y
1174
1175    r = np.abs(y)
1176    out["rectified"] = r
1177
1178    if envelope and 0 < envelope < nyq:
1179        if not HAVE_SCIPY:
1180            raise ImportError("scipy is required for the envelope filter.")
1181        b, a = sp_signal.butter(2, envelope / nyq, btype="low")
1182        out["envelope"] = np.maximum(sp_signal.filtfilt(b, a, r), 0.0)
1183    else:
1184        out["envelope"] = r
1185
1186    w = max(1, int(round(rms_window * fs / 1000.0)))
1187    out["rms"] = np.sqrt(_movmean(y ** 2, w))
1188
1189    if mvc:

1191
1192    out["warnings"] = _emg_warn(out["quality"])
1193    return out

1196def _movmean(x, w):
1197    if w <= 1:
1198        return x.copy()
1199    c = np.concatenate([[0.0], np.cumsum(x)])
1200    n = x.size
1201    half = w // 2
1202    lo = np.clip(np.arange(n) - half, 0, n)
1203    hi = np.clip(np.arange(n) + half + 1, 0, n)
1204    return (c[hi] - c[lo]) / (hi - lo)

1207def _emg_quality(x, fs, notch_hz):
1208    n = x.size
1209    lo, hi = float(np.min(x)), float(np.max(x))
1210    rng = hi - lo
1211    q = {"range": rng, "std": float(np.std(x))}
1212    q["flat"] = rng < np.finfo(float).eps or q["std"] < np.finfo(float).eps
1213
1214    if rng > 0:
1215        tol = 0.01 * rng
1216        q["frac_at_rail"] = float(np.mean((x >= hi - tol) | (x <= lo + tol)))
1217    else:
1218        q["frac_at_rail"] = 1.0
1219    q["saturated"] = q["frac_at_rail"] > 0.20
1220
1221    e = _movmean(np.abs(x - np.median(x)), max(1, int(round(0.05 * fs))))
1222    es = np.sort(e)
1223    baseline = np.median(es[:max(1, n // 10)])
1224    peak = np.median(es[max(1, int(0.9 * n)):])
1225    q["snr_estimate"] = float(peak / baseline) if baseline > 0 else float("inf")
1226
1227    q["mains_ratio"] = float("nan")
1228    f0 = notch_hz if notch_hz else 60.0
1229    if f0 < fs / 2 and n >= 4:
1230        nfft = int(2 ** np.ceil(np.log2(min(n, int(4 * fs)))))
1231        seg = x[:min(n, nfft)]
1232        seg = seg - np.mean(seg)
1233        P = np.abs(np.fft.fft(seg, nfft)) ** 2
1234        fax = np.arange(nfft) * fs / nfft
1235
1236        def band_power(a, b):
1237            m = (fax >= a) & (fax < b)
1238            return float(np.mean(P[m])) if m.any() else 0.0
1239
1240        at = band_power(f0 - 1, f0 + 1)
1241        near = band_power(f0 - 10, f0 - 2) + band_power(f0 + 2, f0 + 10)
1242        if near > 0:
1243            q["mains_ratio"] = at / (near / 2)
1244    return q
